    pub indent: IndentMode,
    pub sort_keys: bool,

pub fn minify(text: &str)                  -> Result<String, JsonitaError>;

2.2实现

pub fn format(text: &str, opts: FormatOpts) -> Result<String, JsonitaError> {
    // 1. 解析为 Value（serde_json 已 preserve insertion order，需要开启 feature）
    //    Cargo.toml: serde_json = { version = "1", features = ["preserve_order"] }
    let mut v: Value = serde_json::from_str(text)
        .map_err(error_loc::map)?;

    // 2. sort_keys（递归）
    if opts.sort_keys { sort_keys_recursive(&mut v); }

    // 3. 用 PrettyFormatter 输出
    let indent: &[u8] = match opts.indent {
        IndentMode::Spaces2 => b"  ",
        IndentMode::Spaces4 => b"    ",
        IndentMode::Tab     => b"\t",
    };

    let formatter = serde_json::ser::PrettyFormatter::with_indent(indent);
    let mut buf = Vec::with_capacity(text.len() + 64);
    let mut ser = serde_json::Serializer::with_formatter(&mut buf, formatter);
    use serde::Serialize;
    v.serialize(&mut ser).map_err(|e| JsonitaError::Io(e.to_string()))?;

    let mut out = String::from_utf8(buf)
        .map_err(|e| JsonitaError::Io(e.to_string()))?;
    if opts.trailing_newline && !out.ends_with('\n') { out.push('\n'); }
    Ok(out)
}

pub fn minify(text: &str) -> Result<String, JsonitaError> {
    let v: Value = serde_json::from_str(text).map_err(error_loc::map)?;
    serde_json::to_string(&v).map_err(|e| JsonitaError::Io(e.to_string()))
}

fn sort_keys_recursive(v: &mut Value) {
    match v {
        Value::Object(map) => {
            // serde_json::Map (preserve_order) 用 IndexMap，sort 需重构
            let mut keys: Vec<String> = map.keys().cloned().collect();
            keys.sort();
            let mut new_map = serde_json::Map::with_capacity(keys.len());
            for k in keys {
                let mut val = map.remove(&k).unwrap();
                sort_keys_recursive(&mut val);
                new_map.insert(k, val);
            }
            *map = new_map;
        }
        Value::Array(arr) => arr.iter_mut().for_each(sort_keys_recursive),
        _ => {}
    }
}

3错误位置定位

3.1策略

serde_json 的 Error 提供 line() 和 column()（基于 deserializer 状态），但这俩 1-indexed 且与用户看到的位置一致。提取要点：

// src-tauri/src/engine/error_loc.rs
use serde_json::Error as SjErr;
use crate::error::JsonitaError;

pub fn map(e: SjErr) -> JsonitaError {
    JsonitaError::Parse {
        line: e.line() as u32,
        col:  e.column() as u32,
        msg:  classify(&e),
    }
}

fn classify(e: &SjErr) -> String {
    // serde_json 错误信息形如 "expected `,` or `}` at line 3 column 5"
    // 取冒号前的描述部分给用户看，去掉 "at line ... column ..." 后缀
    let raw = e.to_string();
    raw.split(" at line ").next().unwrap_or(&raw).to_string()
}

3.2常见错误分类

trailing comma"trailing comma not allowed"
key must be a string"key must be a string (use double quotes)"

4嵌套 stringified JSON 全量解开

4.1问题定义

来自 plan/01 F3.3：Go proto 序列化 / 网关包裹常见 ── JSON 字段值本身又是 stringified JSON。一键递归解开：

// before
{
  "code": 200,
  "data": "{\"name\":\"alice\"}",
  "extra": "{\"tags\":[\"a\",\"b\"]}"
}

// after
{
  "code": 200,
  "data": { "name": "alice" },
  "extra": { "tags": ["a", "b"] }
}

4.2算法

// src-tauri/src/engine/unwrap.rs
use serde_json::Value;
use std::time::{Duration, Instant};
use crate::error::JsonitaError;

pub struct UnwrapOpts {
    pub timeout_ms: u64,                   // 默认 200
    pub max_depth: Option<u32>,         // None = 无限制
}

/// 递归解开嵌套 stringified JSON。
/// 策略：对每个 string 值尝试 JSON.parse；成功且结果是 object/array 则替换。
/// 终止条件：超时 / 达到 max_depth / 不再有可解的 string。
pub fn unwrap(text: &str, opts: UnwrapOpts) -> Result<String, JsonitaError> {
    let deadline = Instant::now() + Duration::from_millis(opts.timeout_ms);
    let mut v: Value = serde_json::from_str(text).map_err(super::error_loc::map)?;
    walk(&mut v, deadline, opts.max_depth, 0)?;
    serde_json::to_string_pretty(&v).map_err(|e| JsonitaError::Io(e.to_string()))
}

fn walk(
    v: &mut Value,
    deadline: Instant,
    max_depth: Option<u32>,
    depth: u32,
) -> Result<(), JsonitaError> {
    if Instant::now() >= deadline {
        return Err(JsonitaError::UnwrapTimeout { ms: 0, depth });
    }
    if let Some(md) = max_depth { if depth >= md { return Ok(()); } }

    match v {
        Value::String(s) => {
            // 尝试 parse 当前 string 为 JSON
            if let Ok(parsed) = serde_json::from_str::<Value>(s) {
                if matches!(parsed, Value::Object(_) | Value::Array(_)) {
                    *v = parsed;
                    walk(v, deadline, max_depth, depth + 1)?;
                }
            }
        }
        Value::Object(map) => {
            for (_, child) in map.iter_mut() { walk(child, deadline, max_depth, depth + 1)?; }
        }
        Value::Array(arr) => {
            for child in arr.iter_mut() { walk(child, deadline, max_depth, depth + 1)?; }
        }
        _ => {}
    }
    Ok(())
}

4.3关键决策

决策选择理由
string 是数字 / bool 时是否解不解（只解到 object / array）避免破坏 "123" 这种业务上有意义的字符串
解开后是否再 walk是支持多层嵌套（如 "{\\"a\\":\\"{\\\\\\"b\\\\\\":1}\\"}"）
遇到错误 string静默跳过，保留原 string不是"修复"工具，无效字符串不该报错
超时单位每次 walk 检查 Instant，不是按节点数避免大对象遍历时仍卡死

5.2实现

pub fn json_to_string(text: &str, opts: StringifyOpts) -> Result<String, JsonitaError> {
    // 1. 验证合法 JSON
    let v: Value = serde_json::from_str(text).map_err(super::error_loc::map)?;
    // 2. 转回 string（minify or pretty）
    let raw = if opts.minify {
        serde_json::to_string(&v)
    } else {
        serde_json::to_string_pretty(&v)
    }.map_err(|e| JsonitaError::Io(e.to_string()))?;
    // 3. 按 quote style 包裹 + 转义内部 quote
    let (open, close, esc) = match opts.quote {
        QuoteStyle::Double => ("\"", "\"", '"'),
        QuoteStyle::Single => ("'",  "'",  '\''),
    };
    let mut out = String::with_capacity(raw.len() + 2);
    out.push_str(open);
    for c in raw.chars() {
        match c {
            '\\' => out.push_str("\\\\"),
            c if c == esc => { out.push('\\'); out.push(c); }
            '\n' => out.push_str("\\n"),
            '\t' => out.push_str("\\t"),
            '\r' => out.push_str("\\r"),
            c if opts.escape_unicode && c as u32 > 127 => {
                let mut b = [0u16; 2]; let s = c.encode_utf16(&mut b);
                for unit in s { out.push_str(&format!("\\u{:04X}", unit)); }
            }
            c => out.push(c),
        }
    }
    out.push_str(close);
    Ok(out)
}

pub fn string_to_json(text: &str) -> Result<String, JsonitaError> {
    // 1. 剥去最外层 quote（容忍 single / double）
    let trimmed = text.trim();
    let inner = if (trimmed.starts_with('"')  && trimmed.ends_with('"'))
                 || (trimmed.starts_with('\'') && trimmed.ends_with('\'')) {
        &trimmed[1..trimmed.len()-1]
    } else {
        trimmed   // 已经没有外层包裹，直接当 raw json 处理
    };
    // 2. 反转义
    let unescaped = unescape(inner)?;
    // 3. 验证 + 格式化
    let v: Value = serde_json::from_str(&unescaped).map_err(super::error_loc::map)?;
    serde_json::to_string_pretty(&v).map_err(|e| JsonitaError::Io(e.to_string()))
}

fn unescape(s: &str) -> Result<String, JsonitaError> {
    let mut out = String::with_capacity(s.len());
    let mut chars = s.chars();
    while let Some(c) = chars.next() {
        if c != '\\' { out.push(c); continue; }
        match chars.next() {
            Some('"')  => out.push('"'),
            Some('\'') => out.push('\''),
            Some('\\') => out.push('\\'),
            Some('/')  => out.push('/'),
            Some('n')  => out.push('\n'),
            Some('r')  => out.push('\r'),
            Some('t')  => out.push('\t'),
            Some('b')  => out.push('\u{08}'),
            Some('f')  => out.push('\u{0C}'),
            Some('u') => {
                let hex: String = chars.by_ref().take(4).collect();
                let code = u32::from_str_radix(&hex, 16)
                    .map_err(|_| JsonitaError::Parse { line:0, col:0, msg: format!("invalid \\u{}", hex) })?;
                if let Some(c) = char::from_u32(code) { out.push(c); }
            }
            Some(other) => { out.push('\\'); out.push(other); }
            None => return Err(JsonitaError::Parse { line:0, col:0, msg: "trailing backslash".into() }),
        }
    }
    Ok(out)
}

6"自动解开" 与 format 的组合

plan/01 F7.5 ── 设置 auto_unwrap = true 时，json_format 内部先调 unwrap 再 format。

// src-tauri/src/commands/json_ops.rs
#[tauri::command]
pub async fn json_format(
    text: String,
    opts: FormatOpts,
    settings: tauri::State<'_, SettingsStore>,
) -> Result<String, JsonitaError> {
    let s = settings.get();
    let processed = if s.auto_unwrap {
        engine::unwrap::unwrap(&text, UnwrapOpts {
            timeout_ms: s.unwrap_timeout_ms,
            max_depth:  None,
        })?
    } else {
        text
    };
    tauri::async_runtime::spawn_blocking(move || engine::json::format(&processed, opts))
        .await
        .map_err(|e| JsonitaError::Io(e.to_string()))?
}

7I/O 解耦原则

  • engine::* 所有函数纯（只接 str → 返 String / Error），不读 fs / net / db
  • Tauri command 层负责拼装：读 settings → 调 engine → 写 history
  • 所有 engine 函数可在 cargo test 中无 Tauri 环境跑
