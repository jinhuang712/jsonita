2SQLite schema

2.1表结构

-- migrations/0001_init.sql
PRAGMA journal_mode = WAL;
PRAGMA synchronous  = NORMAL;
PRAGMA foreign_keys = ON;

CREATE TABLE history (
  id            INTEGER PRIMARY KEY AUTOINCREMENT,
  created_at    INTEGER NOT NULL,                 -- unix ms
  content       TEXT    NOT NULL,
  summary       TEXT    NOT NULL,                 -- content 前 80 字符
  content_hash  TEXT    NOT NULL,                 -- sha256 hex
  op_type       TEXT    NOT NULL                  -- 'format' | 'minify' | 'tree' | ...
                CHECK (op_type IN ('format','minify','tree','str-to-json','json-to-str','ai-fix')),
  pinned        INTEGER NOT NULL DEFAULT 0 CHECK (pinned IN (0,1)),
  starred       INTEGER NOT NULL DEFAULT 0 CHECK (starred IN (0,1)),
  tags          TEXT                              -- 预留 JSON array of strings
);

CREATE UNIQUE INDEX idx_history_hash ON history (content_hash);
CREATE INDEX        idx_history_pinned_created ON history (pinned DESC, created_at DESC);
CREATE INDEX        idx_history_starred ON history (starred);

-- FTS5 虚拟表用于搜索
CREATE VIRTUAL TABLE history_fts USING fts5(
  content,
  summary,
  content='history',
  content_rowid='id',
  tokenize = 'unicode61 remove_diacritics 2'
);
-- 触发器同步主表 ↔ FTS
CREATE TRIGGER history_ai AFTER INSERT ON history BEGIN
  INSERT INTO history_fts (rowid, content, summary) VALUES (new.id, new.content, new.summary);
END;
CREATE TRIGGER history_ad AFTER DELETE ON history BEGIN
  INSERT INTO history_fts (history_fts, rowid, content, summary)
    VALUES('delete', old.id, old.content, old.summary);
END;
CREATE TRIGGER history_au AFTER UPDATE ON history BEGIN
  INSERT INTO history_fts (history_fts, rowid, content, summary)
    VALUES('delete', old.id, old.content, old.summary);
  INSERT INTO history_fts (rowid, content, summary) VALUES (new.id, new.content, new.summary);
END;

-- 上次会话（单行）
CREATE TABLE last_session (
  id          INTEGER PRIMARY KEY CHECK (id = 1),  -- 强制单行
  content     TEXT    NOT NULL,
  op_type     TEXT    NOT NULL,
  saved_at    INTEGER NOT NULL                     -- unix ms
);

-- 应用元数据（版本 / 启动次数等）
CREATE TABLE app_meta (
  key   TEXT PRIMARY KEY,
  value TEXT NOT NULL
);
INSERT INTO app_meta (key, value) VALUES
  ('schema_version', '1'),
  ('install_at',     strftime('%s','now') * 1000);

2.2migration 框架

// src-tauri/src/store/db.rs
use rusqlite::{Connection, params};
use std::path::PathBuf;
use crate::error::JsonitaError;

pub struct Db { conn: Connection }

impl Db {
    pub fn open(app: &tauri::AppHandle) -> Result<Self, JsonitaError> {
        let path = db_path(app)?;
        std::fs::create_dir_all(path.parent().unwrap()).ok();
        let mut conn = Connection::open(&path)
            .map_err(|e| JsonitaError::Sqlite(e.to_string()))?;
        migrate(&mut conn)?;
        Ok(Self { conn })
    }
}

const MIGRATIONS: &[(u32, &str)] = &[
    (1, include_str!("migrations/0001_init.sql")),
    // future: (2, include_str!("migrations/0002_xxx.sql")),
];

fn migrate(conn: &mut Connection) -> Result<(), JsonitaError> {
    conn.execute_batch("CREATE TABLE IF NOT EXISTS schema_version (v INTEGER NOT NULL)")
        .map_err(|e| JsonitaError::Sqlite(e.to_string()))?;
    let current: u32 = conn.query_row(
        "SELECT COALESCE(MAX(v), 0) FROM schema_version", [],
        |r| r.get(0),
    ).map_err(|e| JsonitaError::Sqlite(e.to_string()))?;

    let tx = conn.transaction().map_err(|e| JsonitaError::Sqlite(e.to_string()))?;
    for (v, sql) in MIGRATIONS {
        if *v > current {
            tx.execute_batch(sql).map_err(|e| JsonitaError::Sqlite(format!("migration {}: {}", v, e)))?;
            tx.execute("INSERT INTO schema_version (v) VALUES (?1)", params![v])
              .map_err(|e| JsonitaError::Sqlite(e.to_string()))?;
        }
    }
    tx.commit().map_err(|e| JsonitaError::Sqlite(e.to_string()))?;
    Ok(())
}

fn db_path(app: &tauri::AppHandle) -> Result<PathBuf, JsonitaError> {
    app.path().app_data_dir()
        .map(|p| p.join("history.db"))
        .map_err(|e| JsonitaError::Io(e.to_string()))
}

2.3history CRUD

// src-tauri/src/store/history.rs
use rusqlite::params;
use sha2::{Digest, Sha256};
use crate::commands::history::{HistoryRow, OpType, ListOpts};
use crate::error::JsonitaError;

pub fn add(conn: &rusqlite::Connection, content: &str, op_type: OpType,
           settings_limit: u32) -> Result<(), JsonitaError> {
    let hash = sha256_hex(content);
    let summary: String = content.chars().take(80).collect();
    let now = chrono::Utc::now().timestamp_millis();

    // UPSERT 去重（同 hash 已存在则更新 created_at + op_type）
    conn.execute(
        "INSERT INTO history (created_at, content, summary, content_hash, op_type)
         VALUES (?1, ?2, ?3, ?4, ?5)
         ON CONFLICT(content_hash) DO UPDATE SET
            created_at = excluded.created_at,
            op_type    = excluded.op_type",
        params![now, content, summary, hash, op_type.as_str()],
    ).map_err(|e| JsonitaError::Sqlite(e.to_string()))?;

    // 滚动裁剪：保留 pinned + starred + 最近 N 条
    conn.execute(
        "DELETE FROM history
         WHERE id NOT IN (
            SELECT id FROM history
            WHERE pinned = 1 OR starred = 1
            UNION ALL
            SELECT id FROM history
            WHERE pinned = 0 AND starred = 0
            ORDER BY created_at DESC
            LIMIT ?1
         )",
        params![settings_limit],
    ).map_err(|e| JsonitaError::Sqlite(e.to_string()))?;
    Ok(())
}

pub fn list(conn: &rusqlite::Connection, opts: ListOpts) -> Result<Vec<HistoryRow>, JsonitaError> {
    // pinned 优先，其次 created_at DESC
    let mut stmt = conn.prepare(
        "SELECT id, created_at, content, summary, content_hash, op_type, pinned, starred
         FROM history
         WHERE (?1 IS NULL OR pinned   = ?1)
           AND (?2 IS NULL OR starred  = ?2)
         ORDER BY pinned DESC, created_at DESC
         LIMIT ?3 OFFSET ?4",
    ).map_err(|e| JsonitaError::Sqlite(e.to_string()))?;

    let rows = stmt.query_map(
        params![opts.only_pinned, opts.only_starred, opts.limit, opts.offset],
        row_to_history,
    ).map_err(|e| JsonitaError::Sqlite(e.to_string()))?;

    rows.collect::<Result<_,_>>()
        .map_err(|e| JsonitaError::Sqlite(e.to_string()))
}

pub fn search(conn: &rusqlite::Connection, query: &str, limit: u32)
    -> Result<Vec<HistoryRow>, JsonitaError> {
    let mut stmt = conn.prepare(
        "SELECT h.id, h.created_at, h.content, h.summary, h.content_hash,
                h.op_type, h.pinned, h.starred
         FROM history_fts f JOIN history h ON h.id = f.rowid
         WHERE history_fts MATCH ?1
         ORDER BY rank
         LIMIT ?2",
    ).map_err(|e| JsonitaError::Sqlite(e.to_string()))?;
    let rows = stmt.query_map(params![query, limit], row_to_history)
        .map_err(|e| JsonitaError::Sqlite(e.to_string()))?;
    rows.collect::<Result<_,_>>()
        .map_err(|e| JsonitaError::Sqlite(e.to_string()))
}

fn sha256_hex(s: &str) -> String {
    let mut h = Sha256::new();
    h.update(s.as_bytes());
    format!("{:x}", h.finalize())
}

2.4last_session CRUD

// src-tauri/src/store/session.rs
use rusqlite::params;
use crate::commands::session::LastSession;
use crate::error::JsonitaError;

pub fn save(conn: &rusqlite::Connection, s: &LastSession) -> Result<(), JsonitaError> {
    conn.execute(
        "INSERT INTO last_session (id, content, op_type, saved_at)
         VALUES (1, ?1, ?2, ?3)
         ON CONFLICT(id) DO UPDATE SET
            content = excluded.content,
            op_type = excluded.op_type,
            saved_at = excluded.saved_at",
        params![s.content, s.op_type.as_str(), s.saved_at],
    ).map_err(|e| JsonitaError::Sqlite(e.to_string()))?;
    Ok(())
}

pub fn load(conn: &rusqlite::Connection) -> Result<Option<LastSession>, JsonitaError> {
    let r = conn.query_row(
        "SELECT content, op_type, saved_at FROM last_session WHERE id = 1",
        [],
        |row| Ok(LastSession {
            content:  row.get(0)?,
            op_type:  OpType::from_str(&row.get::<_, String>(1)?).unwrap_or(OpType::Format),
            saved_at: row.get(2)?,
        }),
    );
    match r {
        Ok(s) => Ok(Some(s)),
        Err(rusqlite::Error::QueryReturnedNoRows) => Ok(None),
        Err(e) => Err(JsonitaError::Sqlite(e.to_string())),
    }
}

pub fn clear(conn: &rusqlite::Connection) -> Result<(), JsonitaError> {
    conn.execute("DELETE FROM last_session WHERE id = 1", [])
        .map_err(|e| JsonitaError::Sqlite(e.to_string()))?;
    Ok(())
}

3连接池（rusqlite + r2d2）

// src-tauri/src/store/db.rs (continued)
use r2d2::Pool;
use r2d2_sqlite::SqliteConnectionManager;

pub type DbPool = Pool<SqliteConnectionManager>;

pub fn build_pool(app: &tauri::AppHandle) -> Result<DbPool, JsonitaError> {
    let path = db_path(app)?;
    let mgr = SqliteConnectionManager::file(path)
        .with_init(|c| { c.execute_batch("
            PRAGMA journal_mode = WAL;
            PRAGMA synchronous  = NORMAL;
            PRAGMA foreign_keys = ON;
            PRAGMA busy_timeout = 5000;
        ")?; Ok(()) });
    Pool::builder()
        .max_size(4)
        .build(mgr)
        .map_err(|e| JsonitaError::Sqlite(e.to_string()))
}

WAL 模式下读 / 写并发安全，max_size=4 已足够（v1 单进程 + 异步运行时几个 worker）。

4Keychain 封装

// src-tauri/src/system/keychain.rs
use keyring::Entry;
use crate::error::JsonitaError;

const SERVICE: &str = "com.jsonita.app";

pub struct KeyStore;

impl KeyStore {
    pub fn set(account: &str, value: &str) -> Result<(), JsonitaError> {
        let entry = Entry::new(SERVICE, account)
            .map_err(|e| JsonitaError::Keychain(e.to_string()))?;
        entry.set_password(value).map_err(|e| JsonitaError::Keychain(e.to_string()))
    }

    pub fn get(account: &str) -> Result<Option<String>, JsonitaError> {
        let entry = Entry::new(SERVICE, account)
            .map_err(|e| JsonitaError::Keychain(e.to_string()))?;
        match entry.get_password() {
            Ok(s) => Ok(Some(s)),
            Err(keyring::Error::NoEntry) => Ok(None),
            Err(e) => Err(JsonitaError::Keychain(e.to_string())),
        }
    }

    pub fn delete(account: &str) -> Result<(), JsonitaError> {
        let entry = Entry::new(SERVICE, account)
            .map_err(|e| JsonitaError::Keychain(e.to_string()))?;
        match entry.delete_credential() {
            Ok(()) => Ok(()),
            Err(keyring::Error::NoEntry) => Ok(()),
            Err(e) => Err(JsonitaError::Keychain(e.to_string())),
        }
    }
}

5设置项 store

用 tauri-plugin-store（KV JSON 文件）。一次性加载到内存，所有 read 走缓存，write 触发 emit 事件。

// src-tauri/src/store/settings.rs
use parking_lot::RwLock;
use std::sync::Arc;
use serde::{Serialize, Deserialize};
use crate::error::JsonitaError;

#[derive(Serialize, Deserialize, Clone)]
#[serde(rename_all = "camelCase", default)]
pub struct Settings { /* 见 01_ipc § 2.1.4 */ }

impl Default for Settings { /* 默认值 */ }

pub struct SettingsStore {
    cache: Arc<RwLock<Settings>>,
    app:   tauri::AppHandle,
}

impl SettingsStore {
    pub fn load(app: &tauri::AppHandle) -> Result<Self, JsonitaError> {
        let store = app.store("settings.json")
            .map_err(|e| JsonitaError::Io(e.to_string()))?;
        let v: Settings = match store.get("settings") {
            Some(v) => serde_json::from_value(v).unwrap_or_default(),
            None    => Settings::default(),
        };
        Ok(Self {
            cache: Arc::new(RwLock::new(v)),
            app:   app.clone(),
        })
    }

    pub fn get(&self) -> Settings { self.cache.read().clone() }

    pub fn patch(&self, patch: serde_json::Map<String, serde_json::Value>)
        -> Result<Settings, JsonitaError>
    {
        let mut current = self.cache.write();
        let mut as_value = serde_json::to_value(&*current).unwrap();
        let obj = as_value.as_object_mut().unwrap();
        for (k, v) in patch { obj.insert(k, v); }
        let updated: Settings = serde_json::from_value(as_value)
            .map_err(|e| JsonitaError::Io(e.to_string()))?;
        *current = updated.clone();

        // 落盘
        let store = self.app.store("settings.json")
            .map_err(|e| JsonitaError::Io(e.to_string()))?;
        store.set("settings", serde_json::to_value(&updated).unwrap());
        store.save().map_err(|e| JsonitaError::Io(e.to_string()))?;

        // 广播
        let _ = self.app.emit("settings:changed", updated.clone());
        Ok(updated)
    }
}

6.4"窗口期" 内存计时器实现

// src-tauri/src/system/restore_timer.rs
use tokio::sync::Mutex;
use tokio::task::JoinHandle;
use std::sync::Arc;
use std::time::Duration;

pub struct RestoreTimer {
    inner: Arc<Mutex<Option<JoinHandle<()>>>,
}

impl RestoreTimer {
    pub fn new() -> Self { Self { inner: Default::default() } }

    pub async fn arm(&self, ms: u64) {
        let mut lock = self.inner.lock().await;
        if let Some(prev) = lock.take() { prev.abort(); }
        // 注意：我们不需要 timer fire 时做任何事 ──
        // load_last 时通过比较 now - saved_at 即可判断是否过期
        // timer 仅用于内存中 "知道窗口期开始的时刻" 这个抽象
        // 真正的过期判定在 React listen window:shown 时由 Rust 端判断
        let task = tokio::spawn(async move {
            tokio::time::sleep(Duration::from_millis(ms)).await;
        });
        *lock = Some(task);
    }

    pub async fn disarm(&self) {
        let mut lock = self.inner.lock().await;
        if let Some(prev) = lock.take() { prev.abort(); }
    }
}

注意：实际"是否在窗口期内"由 window:shown 触发时 Rust 端读 last_session + 比较时间戳决定，timer 只是显式建模"窗口期生命周期"概念，方便未来扩展（如增加"窗口期内每次呼出延长 timer"等行为）。

9不变量

  • I-1：history.content_hash 唯一 ── 同内容多次操作合并为一条（更新 op_type / created_at）
  • I-2：last_session 至多一行（CHECK (id = 1)）
  • I-3：所有 DB 写在单一事务（rusqlite::Transaction），避免半成品
  • I-4：API key 不出现于任何 fs 文件 / 日志 / event payload
  • I-5：所有 settings.json 读 / 写经过 SettingsStore，不允许其他模块直接打开
