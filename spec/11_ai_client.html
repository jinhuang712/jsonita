AI 客户端

DeepSeek 请求构造 · Prompt 模板 · 响应验证 · 错误透传 · diff 计算 · 测试连接。

1原则锁定

  • 用户用自己 API key ── 不当家长，不限频，不缓存
  • 仅修复，不解释 ── prompt 严格要求只返回合法 JSON
  • 错误透传 ── 上游 429 / 5xx 原样回前端，附 retry-after
  • 开关完全关 ── ai_enabled=false 时入口隐藏 + Rust 端 early return
  • 无 telemetry ── 不把用户 JSON 发到任何 Jsonita 自己的后端

// src-tauri/src/ai/deepseek.rs
use serde::{Serialize, Deserialize};

#[derive(Serialize)]
struct ChatRequest<'a> {
    model:        &'a str,
    messages:     Vec<ChatMessage<'a>>,
    temperature:  f32,
    max_tokens:   u32,
    stream:       bool,
    response_format: ResponseFormat,
}

#[derive(Serialize)]
struct ChatMessage<'a> { role: &'a str, content: &'a str }

#[derive(Serialize)]
#[serde(tag = "type")]
enum ResponseFormat {
    #[serde(rename = "json_object")] JsonObject,
}

#[derive(Deserialize)]
struct ChatResponse {
    choices:  Vec<Choice>,
    usage:    Option<Usage>,
    model:    Option<String>,
}

#[derive(Deserialize)]
struct Choice { message: ChatMessageOut, finish_reason: Option<String> }

#[derive(Deserialize)]
struct ChatMessageOut { content: String }

#[derive(Deserialize)]
struct Usage { prompt_tokens: u32, completion_tokens: u32 }

4Prompt 模板

// src-tauri/src/ai/prompt.rs
pub fn system_prompt() -> &'static str {
    r#"You are a JSON repair tool. Your only job is to fix invalid JSON.

RULES:
1. Output a SINGLE valid JSON object or array.
2. Do NOT add any explanation, markdown, code fence, or commentary.
3. Preserve the original intent: keep all keys and values exactly as the user
   typed them, only fix syntax (quotes, commas, brackets, escapes).
4. If the input is already valid JSON, return it unchanged (after standard
   pretty-printing).
5. If repair is impossible, return:
   { "_jsonita_repair_failed": true, "reason": "<short reason>" }

OUTPUT FORMAT: plain JSON text. No prefix, no suffix."#
}

pub fn user_prompt(text: &str, line: Option<u32>, col: Option<u32>, msg: Option<&str>) -> String {
    let mut prompt = String::from("Fix this JSON:\n\n```\n");
    prompt.push_str(text);
    prompt.push_str("\n```\n");
    if let (Some(l), Some(c)) = (line, col) {
        use std::fmt::Write;
        let _ = write!(prompt, "\nHint: parse error at line {}, column {}", l, c);
        if let Some(m) = msg { let _ = write!(prompt, " ({})", m); }
        prompt.push('.');
    }
    prompt
}

5请求执行

// src-tauri/src/ai/deepseek.rs (continued)
use crate::error::JsonitaError;
use crate::system::keychain::KeyStore;
use crate::ai::{prompt, validate};
use reqwest::Client;
use std::time::{Duration, Instant};
use tauri::{AppHandle, Manager};
use uuid::Uuid;

#[derive(Serialize, Clone)]
#[serde(rename_all = "camelCase", tag = "phase", content = "data")]
pub enum AiProgress {
    Started,
    Waiting { elapsed_ms: u64 },
    Received { tokens: u32 },
    Validating,
    Done,
}

pub async fn fix(
    app: &AppHandle,
    req: AiFixReq,
) -> Result<AiFixResp, JsonitaError> {
    let started = Instant::now();
    let _ = app.emit("ai:progress", AiProgress::Started);

    // 1. settings 检查（双重保险）
    let s = app.state::<SettingsStore>().get();
    if !s.ai_enabled {
        return Err(JsonitaError::Io("ai disabled".into()));
    }

    // 2. 取 key
    let key = KeyStore::get("deepseek_api_key")?
        .ok_or_else(|| JsonitaError::Keychain("no api key".into()))?;

    // 3. 构造 body
    let body = ChatRequest {
        model: &s.ai_model_id,
        messages: vec![
            ChatMessage { role: "system", content: prompt::system_prompt() },
            ChatMessage { role: "user",
                          content: &prompt::user_prompt(&req.text, req.error_line, req.error_col,
                                                       req.error_msg.as_deref()) },
        ],
        temperature: 0.0,
        max_tokens:  pick_max_tokens(req.text.len()),
        stream: false,
        response_format: ResponseFormat::JsonObject,
    };

    let app_clone = app.clone();
    // 4. 异步 keepalive emit
    let progress_handle = spawn_progress_emitter(app.clone(), started);

    // 5. HTTP
    let client = Client::builder()
        .timeout(Duration::from_secs(60))
        .build()
        .map_err(|e| JsonitaError::Http { status: 0, body: e.to_string() })?;
    let resp = client
        .post(format!("{}{}", DEEPSEEK_BASE_URL, DEEPSEEK_CHAT_PATH))
        .bearer_auth(&key)
        .json(&body)
        .send()
        .await
        .map_err(|e| JsonitaError::Http { status: 0, body: e.to_string() })?;

    progress_handle.abort();

    // 6. 错误状态码
    let status = resp.status().as_u16();
    if status == 429 {
        let ra = resp.headers().get("retry-after")
            .and_then(|h| h.to_str().ok())
            .and_then(|s| s.parse::<u64>().ok())
            .unwrap_or(60);
        return Err(JsonitaError::RateLimit { retry_after_sec: ra });
    }
    if !resp.status().is_success() {
        let body = resp.text().await.unwrap_or_default();
        return Err(JsonitaError::Http { status, body });
    }

    // 7. parse
    let _ = app_clone.emit("ai:progress", AiProgress::Validating);
    let parsed: ChatResponse = resp.json().await
        .map_err(|e| JsonitaError::Http { status: 0, body: e.to_string() })?;
    let content = parsed.choices.first()
        .map(|c| c.message.content.clone())
        .unwrap_or_default();

    // 8. 验证
    let fixed = validate::extract_json(&content)
        .ok_or(JsonitaError::AiInvalidJson { raw: content.clone() })?;
    let _: serde_json::Value = serde_json::from_str(&fixed)
        .map_err(|_| JsonitaError::AiInvalidJson { raw: content })?;

    let _ = app_clone.emit("ai:progress", AiProgress::Done);

    Ok(AiFixResp {
        fixed,
        model:      parsed.model.unwrap_or_default(),
        tokens_in:  parsed.usage.as_ref().map(|u| u.prompt_tokens).unwrap_or(0),
        tokens_out: parsed.usage.as_ref().map(|u| u.completion_tokens).unwrap_or(0),
        elapsed_ms: started.elapsed().as_millis() as u64,
    })
}

fn pick_max_tokens(input_len: usize) -> u32 {
    let approx = (input_len as u32 / 3) * 2;   // 字符数 ÷ 3 ≈ tokens；× 2 留余量
    approx.clamp(512, 8192)
}

fn spawn_progress_emitter(app: AppHandle, started: Instant) -> tokio::task::JoinHandle<()> {
    tokio::spawn(async move {
        let mut ticker = tokio::time::interval(Duration::from_millis(500));
        ticker.tick().await; // 跳过首个立即 tick
        loop {
            ticker.tick().await;
            let _ = app.emit("ai:progress",
                AiProgress::Waiting { elapsed_ms: started.elapsed().as_millis() as u64 });
        }
    })
}

6响应验证

// src-tauri/src/ai/validate.rs

/// 从 AI 输出中抽出 JSON 字符串（即使模型违反指令包了 markdown 也尽量救回）
pub fn extract_json(raw: &str) -> Option<String> {
    let t = raw.trim();

    // case 1: 直接是 JSON
    if t.starts_with('{') || t.starts_with('[') {
        return Some(t.to_string());
    }

    // case 2: 包了 ```json ... ```（模型违反指令时常见）
    if let Some(start) = t.find("```") {
        let after = &t[start+3..];
        let after = after.trim_start_matches("json").trim_start();
        if let Some(end) = after.rfind("```") {
            let inner = after[..end].trim();
            if inner.starts_with('{') || inner.starts_with('[') {
                return Some(inner.to_string());
            }
        }
    }

    // case 3: 文本中找首个 { 到末尾 } 之间
    let first = t.find(|c: char| c == '{' || c == '[')?;
    let last  = t.rfind(|c: char| c == '}' || c == ']')?;
    if last > first {
        return Some(t[first..=last].to_string());
    }
    None
}

7Diff 计算（前端）

// src/panes/diff.ts
import { diffLines } from 'diff';      // npm: diff ^7.0.0

export type DiffLine = { type: 'eq' | 'add' | 'del'; text: string };

export function computeDiff(before: string, after: string): DiffLine[] {
  const parts = diffLines(before, after, { newlineIsToken: false });
  const out: DiffLine[] = [];
  for (const p of parts) {
    const lines = p.value.split(/\n/);
    if (lines[lines.length - 1] === '') lines.pop();
    for (const line of lines) {
      out.push({
        type: p.added ? 'add' : p.removed ? 'del' : 'eq',
        text: line,
      });
    }
  }
  return out;
}

7.1DiffView 渲染

// src/panes/DiffView.tsx (节选)
import { useMemo } from 'react';
import { computeDiff } from './diff';

interface DiffViewProps {
  before: string;
  after:  string;
  onAccept: () => void;
  onReject: () => void;
}

export function DiffView({ before, after, onAccept, onReject }: DiffViewProps) {
  const lines = useMemo(() => computeDiff(before, after), [before, after]);
  return (
    <div className="diff-view">
      <pre className="diff-content">
        {lines.map((l, i) => (
          <div key={i} className={`diff-line diff-${l.type}`}>
            <span className="diff-marker">
              {l.type === 'add' ? '+' : l.type === 'del' ? '-' : ' '}
            </span>
            <span>{l.text}</span>
          </div>
        ))}
      </pre>
      <div className="diff-actions">
        <Button variant="outline" onClick={onReject}>Cancel</Button>
        <Button onClick={onAccept}>Accept Fix</Button>
      </div>
    </div>
  );
}

10前端 store

// src/store/ai.ts
import { create } from 'zustand';
import { aiFix } from '@/ipc/commands';
import { on } from '@/ipc/events';
import { v4 as uuid } from 'uuid';

interface AiStore {
  status:  'idle' | 'requesting' | 'awaiting-decision';
  before:  string;
  after:   string;
  elapsedMs: number;
  error?:  JsonitaError;
  startFix: (text: string, errLoc: EditorError | null) => Promise<void>;
  accept:   () => void;
  reject:   () => void;
}

export const aiStore = create<AiStore>((set, get) => {
  // listen 进度
  on('ai:progress', (p) => {
    if (p.phase === 'Waiting') set({ elapsedMs: p.data.elapsedMs });
  });
  return {
    status: 'idle', before: '', after: '', elapsedMs: 0,
    async startFix(text, errLoc) {
      set({ status: 'requesting', before: text, after: '', elapsedMs: 0, error: undefined });
      try {
        const resp = await aiFix({
          text,
          errorLine: errLoc?.line,
          errorCol:  errLoc?.col,
          errorMsg:  errLoc?.msg,
          requestId: uuid(),
        });
        set({ status: 'awaiting-decision', after: resp.fixed });
      } catch (e: any) {
        set({ status: 'idle', error: e });
        renderErrorToast(e);
      }
    },
    accept() {
      const { after } = get();
      editorStore.getState().setContent(after);
      historyStore.getState().add({ content: after, opType: 'ai-fix' });
      set({ status: 'idle' });
    },
    reject() { set({ status: 'idle' }); },
  };
});
