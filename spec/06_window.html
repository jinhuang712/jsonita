3NSPanel-like 行为（macOS）

NSWindow 强制 canBecomeKey() 时会抢系统焦点（切应用），用户的 ⌘Tab 切换其他 App 会让浮窗隐没。需要把 NSWindow 替换为 NSPanel + 关键标志位。

3.1实现方案：cocoa crate 直调

// src-tauri/src/system/window_panel.rs
//
// 在 Tauri WebviewWindow 创建后立即调用 promote_to_nspanel()，
// 把底层 NSWindow 替换为 NSPanel + 设置以下标志：
//
//   styleMask:           [.nonactivatingPanel, .titled, .fullSizeContentView]
//   level:               .floating
//   collectionBehavior:  [.canJoinAllSpaces,
//                         .stationary,
//                         .fullScreenAuxiliary,
//                         .ignoresCycle]
//   isMovable:           true
//   isMovableByWindowBackground: false
//
// .nonactivatingPanel 是关键：让 panel 接受点击但不切换应用焦点

#[cfg(target_os = "macos")]
pub fn promote_to_nspanel(window: &tauri::WebviewWindow) -> Result<(), JsonitaError> {
    use cocoa::appkit::{
        NSWindow, NSWindowStyleMask, NSWindowCollectionBehavior,
    };
    use cocoa::base::id;
    use objc::{msg_send, sel, sel_impl};

    let ns_window = window.ns_window()? as id;
    unsafe {
        // 设置 style mask 添加 nonactivatingPanel (1 << 7)
        let mask = NSWindowStyleMask::NSTitledWindowMask
            | NSWindowStyleMask::NSFullSizeContentViewWindowMask
            | NSWindowStyleMask::from_bits_unchecked(1 << 7);
        ns_window.setStyleMask_(mask);

        // collection behavior
        let behavior =
              NSWindowCollectionBehavior::NSWindowCollectionBehaviorCanJoinAllSpaces
            | NSWindowCollectionBehavior::NSWindowCollectionBehaviorStationary
            | NSWindowCollectionBehavior::NSWindowCollectionBehaviorFullScreenAuxiliary;
        ns_window.setCollectionBehavior_(behavior);

        // 提升到 floating level
        let _: () = msg_send![ns_window, setLevel: 3i64];  // NSFloatingWindowLevel
    }
    Ok(())
}

v1 直接维护此小段 unsafe；v2 可换 tauri-nspanel 社区插件（一行替换）。

3.2Windows 端的对应行为

Windows 10/11 不需要等价 NSPanel：

  • "alwaysOnTop": true + "focus": false 已能做到始终置顶
  • ShowWindow(hwnd, SW_SHOWNOACTIVATE) 显示不抢焦点
  • 无需 unsafe 调 cocoa

4启动与预热

为了呼出 < 500 ms，主进程启动时就创建好 WebView 窗口并 hide，后续 window.show() 是亚毫秒级。

// src-tauri/src/main.rs（节选）
fn main() {
    tauri::Builder::default()
        .setup(|app| {
            // 1. 初始化基础设施
            store::db::init(app.handle())?;
            system::tray::build(app.handle())?;
            system::shortcuts::register_defaults(app.handle())?;

            // 2. 创建主浮窗（visible: false，已在 conf 设）
            let win = WebviewWindowBuilder::from_config(
                app.handle(),
                app.config().app.windows.iter()
                    .find(|w| w.label == "main").unwrap().clone(),
            )?.build()?;

            // 3. 应用 NSPanel-like
            #[cfg(target_os = "macos")]
            system::window_panel::promote_to_nspanel(&win)?;

            // 4. 注册关闭拦截
            register_close_intercept(&win);

            // 5. macOS：从 Dock 隐藏（菜单栏 only）── 可选，需要用户设置
            #[cfg(target_os = "macos")]
            if app.state::<Settings>().hide_dock_icon {
                app.set_activation_policy(tauri::ActivationPolicy::Accessory);
            }

            Ok(())
        })
        .invoke_handler(tauri::generate_handler![/* all commands */])
        .run(tauri::generate_context!())
        .expect("error running Jsonita");
}

5.2具体 API

// src-tauri/src/system/cursor.rs
use tauri::{Manager, Monitor};

pub fn locate_window(app: &tauri::AppHandle, label: &str) -> Result<(), JsonitaError> {
    let window = app.get_webview_window(label)
        .ok_or_else(|| JsonitaError::Io("window not found".into()))?;

    let cursor = app.cursor_position()?;     // Tauri 2.x API
    let monitor = app.available_monitors()?
        .into_iter()
        .find(|m| {
            let p = m.position();
            let s = m.size();
            cursor.x >= p.x as f64
                && cursor.x < (p.x + s.width as i32) as f64
                && cursor.y >= p.y as f64
                && cursor.y < (p.y + s.height as i32) as f64
        })
        .or_else(|| app.primary_monitor().ok().flatten())
        .ok_or_else(|| JsonitaError::Io("no monitor".into()))?;

    let win_size = window.outer_size()?;
    let scale = monitor.scale_factor();

    let mx = monitor.position().x as f64 / scale;
    let my = monitor.position().y as f64 / scale;
    let mw = monitor.size().width  as f64 / scale;
    let mh = monitor.size().height as f64 / scale;

    let ww = win_size.width  as f64 / scale;
    let wh = win_size.height as f64 / scale;

    let x = mx + (mw - ww) / 2.0;
    let y = my + (mh - wh) / 3.0;            // 上 1/3

    window.set_position(tauri::LogicalPosition::new(x, y))?;
    Ok(())
}

6失焦自动隐藏

6.1逻辑

// src-tauri/src/system/window_panel.rs (continued)
pub fn install_blur_hide(win: &tauri::WebviewWindow, app: &tauri::AppHandle) {
    let app_handle = app.clone();
    let win_label  = win.label().to_string();

    win.on_window_event(move |event| {
        if let tauri::WindowEvent::Focused(false) = event {
            let settings = app_handle.state::<SettingsStore>();
            if !settings.get().hide_on_blur { return; }

            // 用户可能切到设置 modal 或 history modal ──
            // 这俩是同一窗口内的 React modal，不会失焦；
            // 失焦一定是切了别的应用 → hide
            if let Some(w) = app_handle.get_webview_window(&win_label) {
                let _ = w.hide();
                let _ = app_handle.emit_to(&win_label, "window:hidden", ());
            }
        }
    });
}

6.2例外

场景是否触发隐藏
用户切到其他 App是（hide_on_blur=true 时）
用户打开 React 内的 Settings / History modal否（同窗口，不失焦）
用户右键唤起系统 menu否（macOS NSMenu 不触发 WindowEvent::Focused(false)）
用户 ⌘Tab 切到自己另一个 macOS Space取决于 NSPanel collectionBehavior（已加 canJoinAllSpaces）
系统弹出系统级 dialog（权限请求）有可能触发，需 settings hide_on_blur=false 兜底

7.2拦截 close event

fn register_close_intercept(win: &tauri::WebviewWindow) {
    let win_clone = win.clone();
    win.on_window_event(move |event| {
        if let tauri::WindowEvent::CloseRequested { api, .. } = event {
            api.prevent_close();          // 阻止真关闭
            let _ = win_clone.hide();
        }
    });
}

9窗口大小持久化 & 智能宽度

9.3智能宽度算法（关键）

// 输入：内容指标 + 设置项 + 当前窗口状态
// 输出：理想宽度（u32 px）。若与 current 相同则 no-op。

fn compute_ideal_width(
    metrics:    &ContentMetrics,
    settings:   &Settings,
    state:      &WindowState,        // saved_*, user_dragged
    screen_w:   u32,
    current_w:  u32,
) -> u32 {
    // ① 用户已拖动 → 锁定，不自动扩
    if state.user_dragged { return current_w; }

    // ② soft-wrap 开启 → 无需扩宽（已经换行不溢出）
    if metrics.soft_wrap_on { return current_w; }

    // ③ 用户在设置里关了"智能扩宽"
    if !settings.smart_width { return current_w; }

    // ④ 阈值：行长 ≤ 80 字符不触发
    if metrics.max_line_chars <= 80 { return current_w; }

    // ⑤ 估算 ── 8 px/字符（mono 13px 平均）+ 64 px padding（行号+左右）
    let needed = metrics.max_line_chars * 8 + 64;

    // ⑥ 区间：[720, min(1400, screen_w × 70%)]
    let max = (screen_w as f32 * 0.7).min(1400.0) as u32;
    needed.clamp(720, max)
}

fn resolve_initial_size(state: &WindowState, settings: &Settings) -> (u32, u32) {
    if let (Some(w), Some(h)) = (state.saved_width, state.saved_height) {
        (w, h)                    // 沿用上次（含用户拖动）
    } else {
        (settings.initial_width,  // 默认 920，用户可选 720 / 860 / 920 / 1080
         560)                      // 高度首版固定
    }
}
